cmake_minimum_required(VERSION 3.16)
project(CoolingRelief VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets SerialPort)
add_library(relief_ui STATIC mainwindow.cpp mainwindow.h archive.cpp archive.h
    protocol.cpp protocol.h curveplot.cpp curveplot.h)
target_link_libraries(relief_ui PUBLIC Qt6::Widgets Qt6::SerialPort)
target_include_directories(relief_ui PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
qt_add_executable(CoolingRelief WIN32 main.cpp)
target_link_libraries(CoolingRelief PRIVATE relief_ui)
include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
if(MSVC)
    target_compile_options(relief_ui PRIVATE /utf-8)
    target_compile_options(CoolingRelief PRIVATE /utf-8)
endif()
